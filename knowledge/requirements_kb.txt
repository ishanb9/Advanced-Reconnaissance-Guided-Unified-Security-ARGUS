# ─── Knowledge Base RAG Dependencies v2 ───────────────────────────────────────
# Install once on your Kali machine, separate from the main platform requirements.
#
# Usage:
#   pip install -r knowledge/requirements_kb.txt
#
# These are optional — the platform runs normally if not installed,
# KB calls will simply return empty strings and the LLM uses its base knowledge.
#
# Full install (recommended):
#   pip install -r knowledge/requirements_kb.txt
#
# Minimal install (no reranking, no YAML support):
#   pip install chromadb sentence-transformers pypdf beautifulsoup4 tqdm lxml

# ── Core (required) ─────────────────────────────────────────────────────────────

# Vector store — persistent, fully local, no cloud
chromadb>=0.5.0

# Embedding model + CrossEncoder reranker
# Downloads all-MiniLM-L6-v2 (~80 MB) on first use, then cached at ~/.cache/huggingface/
# CrossEncoder reranker: cross-encoder/ms-marco-MiniLM-L-6-v2 (~70 MB)
sentence-transformers>=2.7.0

# PDF text extraction
pypdf>=3.0.0

# HTML/MHTML parsing
beautifulsoup4>=4.12.0
lxml>=5.0.0

# Progress bars for bulk ingest
tqdm>=4.66.0

# ── Optional (recommended) ───────────────────────────────────────────────────────

# YAML file support (for custom tip collections in YAML format)
PyYAML>=6.0.0

# ── Notes ────────────────────────────────────────────────────────────────────────
#
# Embedding models (set KB_EMBED_MODEL env var to override):
#   "all-MiniLM-L6-v2"      → ~80 MB,  fast CPU,    good quality  (default)
#   "all-mpnet-base-v2"     → ~420 MB, slower CPU,  better quality
#   "BAAI/bge-small-en-v1.5" → ~130 MB, fast CPU,   good quality
#
# Reranking (set KB_RERANK_MODEL="" to disable):
#   "cross-encoder/ms-marco-MiniLM-L-6-v2"  → ~70 MB, CPU-friendly (default)
#   Set KB_RERANK_MODEL="" to disable reranking for faster searches
#
# Example with higher quality models:
#   KB_EMBED_MODEL=all-mpnet-base-v2 KB_RERANK_MODEL="" python3 ingest_data.py
#
# No GPU required. All models run on CPU.
# Python >= 3.9 required (platform already uses 3.12+)
# ChromaDB stores data at knowledge/chroma_db/ (survives restarts)
